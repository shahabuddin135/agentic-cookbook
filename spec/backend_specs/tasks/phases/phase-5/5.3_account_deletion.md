    from uuid import uuid4
    from fastapi import Response
    from app.models.compliance import DataDeletionRequest

    @router.delete("", status_code=202)
    async def request_deletion(
        current_user: dict = Depends(get_current_user),
        db: AsyncSessionDep,
    ):
        user_id = current_user["user_id"]

        req = DataDeletionRequest(
            id=str(uuid4()),
            user_id=user_id,
            status="pending",
        )
        db.add(req)
        await db.commit()

        return {"message": "Deletion scheduled. Account will be removed within 30 days."}

  important:
    - "Returns 202 Accepted — NOT 200 or 204"
    - "Does NOT delete the user immediately — creates a queue entry only"
    - "Actual deletion is an out-of-scope background process or manual admin operation"
    - "DataDeletionRequest.status starts as 'pending'"
    - "Multiple deletion requests from the same user are allowed (idempotent-ish)"

  acceptance_criteria:
