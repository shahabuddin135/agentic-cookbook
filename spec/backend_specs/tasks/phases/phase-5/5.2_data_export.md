@block TASK task_5_2
priority: high
intent: "GDPR Art. 20 data portability — GET /api/profile/export returns JSON file"
scope: phase-5

content:
  file: "backend/app/routers/profile.py"

  implement: |
    import json
    from datetime import datetime
    from fastapi.responses import StreamingResponse
    from sqlmodel import select
    from app.models.conversation import Conversation
    from app.models.message import Message

    @router.get("/export")
    async def export_data(
        current_user: dict = Depends(get_current_user),
        db: AsyncSessionDep,
    ):
        user_id = current_user["user_id"]

        # Fetch user
        user = await db.get(User, user_id)

        # Fetch all conversations
        conv_stmt = select(Conversation).where(Conversation.user_id == user_id)
        convs = (await db.exec(conv_stmt)).all()

        # Fetch all messages for each conversation
        conversations_export = []
        for conv in convs:
            msg_stmt = select(Message).where(
                Message.conversation_id == conv.id
            ).order_by(Message.created_at.asc())
            messages = (await db.exec(msg_stmt)).all()
            conversations_export.append({
                **conv.model_dump(),
                "messages": [m.model_dump() for m in messages],
            })

        export_data = {
            "exported_at": datetime.utcnow().isoformat(),
            "user": {"id": user.id, "name": user.name, "email": user.email,
                     "created_at": user.created_at.isoformat()},
            "conversations": conversations_export,
        }

        json_bytes = json.dumps(export_data, default=str).encode("utf-8")

        return StreamingResponse(
            iter([json_bytes]),
            media_type="application/json",
            headers={"Content-Disposition": "attachment; filename=my-data.json"},
        )

  note: >
    Route must be declared BEFORE the /{conversation_id} style routes in FastAPI
    to avoid path shadowing. In profile router, declare /export before / is fine
    since FastAPI uses explicit matching.

  acceptance_criteria:
    - "GET /api/profile/export returns HTTP 200"
    - "Response header: Content-Disposition: attachment; filename=my-data.json"
    - "Response body is valid JSON"
    - "JSON contains user, conversations, and nested messages"
    - "Endpoint returns 403 without JWT"
@end
