# Task 2.1 — JWKS Fetch + Cache

```slc
@block TASK task_2_1
priority: critical
intent: "Implement JWKS fetch with 1-hour in-memory cache using httpx"
scope: phase-2
depends_on: ["1.3"]
estimate_minutes: 10

content:
  file: "backend/app/auth/jwt_verify.py"

  implement: |
    from datetime import datetime, timedelta
    import httpx
    from app.config import settings

    JWKS_TTL = timedelta(hours=1)
    _jwks_cache: dict | None = None
    _jwks_fetched_at: datetime | None = None

    async def get_jwks() -> dict:
        global _jwks_cache, _jwks_fetched_at

        now = datetime.utcnow()
        if _jwks_cache and _jwks_fetched_at and (now - _jwks_fetched_at) < JWKS_TTL:
            return _jwks_cache

        jwks_url = f"{settings.BETTER_AUTH_URL}/api/auth/jwks"
        async with httpx.AsyncClient(timeout=10.0) as client:
            response = await client.get(jwks_url)
            response.raise_for_status()
            _jwks_cache = response.json()
            _jwks_fetched_at = now

        return _jwks_cache

  boundaries:
    - "httpx timeout=10.0 — prevents request hanging if frontend is down"
    - "TLS verification must remain enabled (no verify=False)"
    - "Cache is module-level (process-scoped) — resets on restart"
    - "raise_for_status() ensures non-200 responses propagate as errors"

  acceptance_criteria:
    - "get_jwks() returns dict with 'keys' array"
    - "Second call within 1 hour does NOT make a new HTTP request"
    - "Call after TTL expiry fetches fresh JWKS"
@end
```
