# Task 4.5 — Phase 4 Smoke Test

```slc
@block TASK task_4_5
priority: high
intent: "Full smoke test of Phase 1-4 stack before GDPR work begins"
scope: phase-4
depends_on: ["4.4"]
estimate_minutes: 15

content:
  prerequisites:
    - "Frontend running on port 3000 (for JWT issuance)"
    - "Backend running on port 8000"
    - "Valid JWT obtained from Better Auth"

  test_cases:
    TC1:
      name: "Health check"
      command: "curl http://localhost:8000/health"
      expected: '{"status": "ok", "version": "1.0.0"}'

    TC2:
      name: "Conversations without auth"
      command: "curl http://localhost:8000/api/conversations"
      expected: "HTTP 403"

    TC3:
      name: "Empty conversations list with auth"
      command: 'curl -H "Authorization: Bearer <JWT>" http://localhost:8000/api/conversations'
      expected: '{"conversations": []}'

    TC4:
      name: "Agent chat creates conversation"
      command: |
        curl -X POST http://localhost:8000/api/agent/chat \
          -H "Authorization: Bearer <JWT>" \
          -H "Content-Type: application/json" \
          -d '{"message": "quick scrambled eggs recipe"}' \
          --no-buffer
      expected: |
        data: {"type": "recipe", "data": {...}, "conversation_id": "uuid", "message_id": "uuid"}
        data: {"type": "done"}

    TC5:
      name: "Check DB state after chat"
      check: "SELECT * FROM conversation WHERE user_id='...' in Neon console"
      expected: "1 row with title from message"

    TC6:
      name: "List conversations after chat"
      expected: '{"conversations": [{"id": "uuid", "title": "quick scrambled..."}]}'

    TC7:
      name: "Get conversation with messages"
      command: "curl -H 'Authorization: Bearer <JWT>' http://localhost:8000/api/conversations/<id>"
      expected: "conversation + 2 messages (user + assistant)"

    TC8:
      name: "Delete conversation"
      command: "curl -X DELETE -H 'Authorization: Bearer <JWT>' http://localhost:8000/api/conversations/<id>"
      expected: "HTTP 204"
      followup: "GET same id → HTTP 404"

  acceptance_criteria:
    - "All 8 test cases pass"
    - "Agent response contains valid recipe JSON (title, ingredients, image)"
    - "DB state matches expected after each operation"
    - "CORS OPTIONS request returns Allow-Origin header"
@end
```
