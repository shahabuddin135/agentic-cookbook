    from agents.mcp import MCPServerSse, MCPServerSseParams
    from app.config import settings

    def create_apify_server() -> MCPServerSse:
        return MCPServerSse(
            params=MCPServerSseParams(
                url="https://mcp.apify.com/?tools=web.harvester/recipes-scraper",
                headers={"Authorization": f"Bearer {settings.APIFY_API_TOKEN}"},
            ),
            cache_tools_list=True,
            name="apify_recipes",
        )

  notes:
    - "SSE = remote transport — no subprocess spawned, requires internet"
    - "cache_tools_list=True avoids a tools-list request on every agent invocation"
    - "name='apify_recipes' — this name is FROZEN (see MEMORY.md)"
    - "The URL includes ?tools= to filter to the recipes-scraper only"

  acceptance_criteria:
    - "create_apify_server() returns MCPServerSse instance without errors"
    - "settings.APIFY_API_TOKEN is used in the Authorization header"
    - "No HTTP requests made until the server's __aenter__ is called"
