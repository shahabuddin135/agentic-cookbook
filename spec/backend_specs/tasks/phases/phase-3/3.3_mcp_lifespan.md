# Task 3.3 — MCP Lifespan Manager

```slc
@block TASK task_3_3
priority: critical
intent: "Start MCP servers on app startup, stop cleanly on shutdown, store on app.state"
scope: phase-3
depends_on: ["3.2", "1.8"]
estimate_minutes: 10

content:
  files:
    - "backend/app/agent/mcp_servers.py (add lifespan function)"
    - "backend/app/main.py (update to use lifespan)"

  lifespan_function: |
    # In backend/app/agent/mcp_servers.py:
    from contextlib import asynccontextmanager

    @asynccontextmanager
    async def lifespan_mcp(app):
        apify = create_apify_server()
        pexels = create_pexels_server()
        connected = []
        try:
            await apify.__aenter__()
            connected.append(apify)
            await pexels.__aenter__()
            connected.append(pexels)
            app.state.mcp_servers = [apify, pexels]
            yield
        finally:
            for server in reversed(connected):
                try:
                    await server.__aexit__(None, None, None)
                except Exception:
                    pass  # Log but don't crash shutdown

  update_main: |
    # In backend/app/main.py — replace the bare FastAPI() call:
    from app.agent.mcp_servers import lifespan_mcp

    app = FastAPI(
        title="Agentic Cookbook API",
        version="1.0.0",
        docs_url="/docs" if settings.DEBUG else None,
        redoc_url=None,
        lifespan=lifespan_mcp,
    )

  acceptance_criteria:
    - "uvicorn startup logs show both MCP servers connected (no errors)"
    - "app.state.mcp_servers is a list of 2 live server instances"
    - "uvicorn shutdown disconnects both MCP servers cleanly"
    - "Server startup failure causes app to abort (not silently continue)"
@end
```
